
# Apollo Client Code Review & Improvements

## Issues Identified

### 1. **Cache Management Problems**

**Current Issues:**
- Manual cache modifications in `useChat.tsx` are overly complex and error-prone
- The subscription update logic has unnecessary complexity with timestamp comparisons
- Cache updates don't properly handle edge cases (duplicates, race conditions)

**Better Approach:**
```typescript
// Use cache policies and proper cache updates
const [sendMessage] = useMutation(SEND_MESSAGE_MUTATION, {
  update(cache, { data }) {
    if (!data?.sendMessage) return;
    
    cache.updateQuery({ query: MESSAGES_QUERY }, (prev) => {
      if (!prev?.messages) return prev;
      
      return {
        messages: {
          ...prev.messages,
          edges: [
            ...prev.messages.edges,
            {
              __typename: 'MessageEdge',
              cursor: data.sendMessage.id,
              node: data.sendMessage,
            }
          ]
        }
      };
    });
  }
});
```

### 2. **Missing Type Safety**

**Current Issues:**
- GraphQL operations lack proper TypeScript integration
- Manual type assertions instead of generated types

**Better Approach:**
```typescript
// Use GraphQL Code Generator with proper typing
import { useMessagesQuery, useSendMessageMutation } from '../generated/graphql';

const { data, loading, error, fetchMore } = useMessagesQuery({
  variables: { first: DEFAULT_PAGE_SIZE },
  notifyOnNetworkStatusChange: true,
});
```

### 3. **Inefficient Subscription Handling**

**Current Issues:**
- Subscription logic duplicates cache update patterns from mutations
- Complex filtering logic that could be handled by Apollo's built-in features

**Better Approach:**
```typescript
// Use subscriptionData properly with cache policies
useSubscription(MESSAGE_UPDATED_SUBSCRIPTION, {
  onData({ data, client }) {
    // Let Apollo handle the cache updates automatically
    // Or use simpler cache modifications
    client.cache.evict({ fieldName: 'messages' });
    client.cache.gc();
  }
});
```

### 4. **Poor Error Handling**

**Current Issues:**
- Basic error handling with toast notifications
- No error boundaries or retry strategies
- Missing network error handling

**Better Approach:**
```typescript
const { data, loading, error, refetch } = useQuery(MESSAGES_QUERY, {
  errorPolicy: 'partial',
  notifyOnNetworkStatusChange: true,
  onError: (error) => {
    if (error.networkError) {
      // Handle network errors differently
    }
    if (error.graphQLErrors?.length) {
      // Handle GraphQL errors
    }
  }
});
```

### 5. **Missing Apollo Client Features**

**What's Missing:**
- No cache policies configuration
- No optimistic UI patterns (except basic optimistic response)
- No proper loading states management
- No cache normalization considerations
- No batch operations
- No cache persistence

```typescript
import { ApolloClient, InMemoryCache, createHttpLink, from } from '@apollo/client';
import { setContext } from '@apollo/client/link/context';
import { onError } from '@apollo/client/link/error';
import { GraphQLWsLink } from '@apollo/client/link/subscriptions';
import { createClient } from 'graphql-ws';
import { getMainDefinition } from '@apollo/client/utilities';
import { split } from '@apollo/client/link/core';
import { relayStylePagination } from '@apollo/client/utilities';

// HTTP Link
const httpLink = createHttpLink({
  uri: 'http://localhost:4000/graphql',
});

// WebSocket Link for subscriptions
const wsLink = new GraphQLWsLink(
  createClient({
    url: 'ws://localhost:4000/graphql',
  })
);

// Split link - use ws for subscriptions, http for queries/mutations
const splitLink = split(
  ({ query }) => {
    const definition = getMainDefinition(query);
    return (
      definition.kind === 'OperationDefinition' &&
      definition.operation === 'subscription'
    );
  },
  wsLink,
  httpLink
);

// Auth link (if needed)
const authLink = setContext((_, { headers }) => {
  const token = localStorage.getItem('token'); // Adjust as needed
  return {
    headers: {
      ...headers,
      authorization: token ? `Bearer ${token}` : '',
    }
  };
});

// Error handling link
const errorLink = onError(({ graphQLErrors, networkError, operation, forward }) => {
  if (graphQLErrors) {
    graphQLErrors.forEach(({ message, locations, path }) =>
      console.error(
        `GraphQL error: Message: ${message}, Location: ${locations}, Path: ${path}`
      )
    );
  }

  if (networkError) {
    console.error(`Network error: ${networkError}`);
    // Handle network errors (retry, redirect to login, etc.)
  }
});

// Enhanced cache with proper type policies
const cache = new InMemoryCache({
  typePolicies: {
    Query: {
      fields: {
        messages: relayStylePagination(['first', 'after']),
      },
    },
    Message: {
      fields: {
        status: {
          merge: (existing, incoming) => incoming,
        },
      },
    },
  },
});

export const client = new ApolloClient({
  link: from([errorLink, authLink, splitLink]),
  cache,
  defaultOptions: {
    watchQuery: {
      errorPolicy: 'partial',
      notifyOnNetworkStatusChange: true,
    },
    query: {
      errorPolicy: 'partial',
    },
  },
});
```

```typescript
import { useState, useCallback } from 'react';
import { NetworkStatus } from '@apollo/client';
import { toast } from 'react-hot-toast';
import {
  useGetMessagesQuery,
  useSendMessageMutation,
  useOnMessageUpdatedSubscription,
  MessageStatus,
  MessageSender,
  GetMessagesDocument,
} from '../__generated__/graphql';

const DEFAULT_PAGE_SIZE = 10;
const PAGINATION_THRESHOLD = 5;

export const useChatHook = () => {
  const [inputText, setInputText] = useState('');

  // Query with proper typing
  const {
    data,
    loading,
    error,
    fetchMore,
    refetch,
    networkStatus,
  } = useGetMessagesQuery({
    variables: {
      first: DEFAULT_PAGE_SIZE,
      after: null,
    },
    notifyOnNetworkStatusChange: true,
    errorPolicy: 'partial',
  });

  // Mutation with better cache handling
  const [sendMessageMutation, { loading: sendingMessage }] = useSendMessageMutation({
    onError: (error) => {
      toast.error(`Failed to send message: ${error.message}`);
    },
    onCompleted: () => {
      setInputText('');
      toast.success('Message sent successfully!');
    },
    update(cache, { data }) {
      if (!data?.sendMessage) return;

      // Use updateQuery for cleaner cache updates
      cache.updateQuery(
        {
          query: GetMessagesDocument,
          variables: { first: DEFAULT_PAGE_SIZE, after: null },
        },
        (prev) => {
          if (!prev?.messages) return prev;

          return {
            messages: {
              ...prev.messages,
              edges: [
                ...prev.messages.edges,
                {
                  __typename: 'MessageEdge' as const,
                  cursor: data.sendMessage.id,
                  node: data.sendMessage,
                },
              ],
            },
          };
        }
      );
    },
    optimisticResponse: (vars) => ({
      sendMessage: {
        __typename: 'Message' as const,
        id: `optimistic-${Date.now()}`,
        text: vars.text,
        status: MessageStatus.Sending,
        updatedAt: new Date().toISOString(),
        sender: MessageSender.Admin,
      },
    }),
  });

  // Subscription with proper handling
  useOnMessageUpdatedSubscription({
    onData({ data: subscriptionData, client }) {
      const message = subscriptionData.data?.messageUpdated;
      if (!message) return;

      // Simple cache update - let Apollo handle the complexity
      client.cache.updateQuery(
        {
          query: GetMessagesDocument,
          variables: { first: DEFAULT_PAGE_SIZE, after: null },
        },
        (prev) => {
          if (!prev?.messages) return prev;

          const existingIndex = prev.messages.edges.findIndex(
            edge => edge.node.id === message.id
          );

          if (existingIndex >= 0) {
            // Update existing message
            const newEdges = [...prev.messages.edges];
            newEdges[existingIndex] = {
              ...newEdges[existingIndex],
              node: message,
            };

            return {
              messages: {
                ...prev.messages,
                edges: newEdges,
              },
            };
          } else {
            // Add new message
            return {
              messages: {
                ...prev.messages,
                edges: [
                  ...prev.messages.edges,
                  {
                    __typename: 'MessageEdge' as const,
                    cursor: message.id,
                    node: message,
                  },
                ],
              },
            };
          }
        }
      );
    },
  });

  // Derived state with proper typing
  const messages = data?.messages?.edges?.map(edge => edge.node) || [];
  const pageInfo = data?.messages?.pageInfo;
  
  // Better loading states
  const isInitialLoading = networkStatus === NetworkStatus.loading;
  const isLoadingMore = networkStatus === NetworkStatus.fetchMore;
  const isRefetching = networkStatus === NetworkStatus.refetch;

  // Improved pagination
  const loadMoreMessages = useCallback(async () => {
    if (!pageInfo?.hasNextPage || isLoadingMore) return;

    try {
      await fetchMore({
        variables: {
          after: pageInfo.endCursor,
        },
      });
    } catch (error) {
      console.error('Failed to load more messages:', error);
      toast.error('Failed to load more messages');
    }
  }, [fetchMore, pageInfo?.hasNextPage, pageInfo?.endCursor, isLoadingMore]);

  const handleEndReached = useCallback(
    (index: number) => {
      if (
        index >= messages.length - PAGINATION_THRESHOLD &&
        pageInfo?.hasNextPage &&
        !isLoadingMore
      ) {
        loadMoreMessages();
      }
    },
    [messages.length, pageInfo?.hasNextPage, isLoadingMore, loadMoreMessages]
  );

  const handleSendMessage = useCallback(async () => {
    const trimmedText = inputText.trim();
    if (!trimmedText) return;

    try {
      await sendMessageMutation({
        variables: { text: trimmedText },
      });
    } catch (error) {
      // Error is handled in onError callback
      console.error('Send message error:', error);
    }
  }, [inputText, sendMessageMutation]);

  const handleRetry = useCallback(() => {
    refetch();
  }, [refetch]);

  return {
    // Data
    messages,
    
    // Loading states
    loading: isInitialLoading,
    isLoadingMore,
    isRefetching,
    sendingMessage,
    
    // Error state
    error,
    
    // Actions
    handleEndReached,
    handleRetry,
    handleSendMessage,
    
    // Input state
    inputText,
    setInputText,
  };
};
```

## Phase 4: Error Boundaries and Better Error Handling

### Step 6: Create Error Boundary

Create `src/components/ErrorBoundary/index.tsx`:
```typescript
import React, { Component, ReactNode } from 'react';
import { ApolloError } from '@apollo/client';

interface Props {
  children: ReactNode;
  fallback?: (error: Error, retry: () => void) => ReactNode;
}

interface State {
  hasError: boolean;
  error: Error | null;
}

export class ErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props);
    this.state = { hasError: false, error: null };
  }

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: any) {
    console.error('Error caught by boundary:', error, errorInfo);
    
    // Send to error reporting service
    if (error instanceof ApolloError) {
      console.error('Apollo Error:', {
        graphQLErrors: error.graphQLErrors,
        networkError: error.networkError,
      });
    }
  }

  handleRetry = () => {
    this.setState({ hasError: false, error: null });
  };

  render() {
    if (this.state.hasError) {
      if (this.props.fallback) {
        return this.props.fallback(this.state.error!, this.handleRetry);
      }

      return (
        <div style={{ padding: '20px', textAlign: 'center' }}>
          <h2>Something went wrong</h2>
          <p>{this.state.error?.message}</p>
          <button onClick={this.handleRetry}>Try Again</button>
        </div>
      );
    }

    return this.props.children;
  }
}
```
